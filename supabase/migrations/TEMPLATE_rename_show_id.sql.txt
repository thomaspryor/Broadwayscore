-- Pre-flight: counts (0 is fine) and the UNIQUE-collision check (must be 0 rows).
SELECT 'reviews'    AS t, count(*) FROM reviews    WHERE show_id = :'old_id'
UNION ALL SELECT 'watchlist',  count(*) FROM watchlist  WHERE show_id = :'old_id'
UNION ALL SELECT 'list_items', count(*) FROM list_items WHERE show_id = :'old_id';

SELECT w.user_id FROM watchlist w WHERE w.show_id = :'old_id'
  AND EXISTS (SELECT 1 FROM watchlist x WHERE x.user_id = w.user_id AND x.show_id = :'new_id');
SELECT li.list_id FROM list_items li WHERE li.show_id = :'old_id'
  AND EXISTS (SELECT 1 FROM list_items x WHERE x.list_id = li.list_id AND x.show_id = :'new_id');

BEGIN;
UPDATE reviews    SET show_id = :'new_id' WHERE show_id = :'old_id';
UPDATE watchlist  SET show_id = :'new_id' WHERE show_id = :'old_id';
UPDATE list_items SET show_id = :'new_id' WHERE show_id = :'old_id';
